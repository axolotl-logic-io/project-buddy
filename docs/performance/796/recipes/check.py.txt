def keys(value,required,optional=set()):
 assert isinstance(value,dict)
 assert set(value)>=required,(required-set(value))
 assert not set(value)-required-optional,('unknown fields',set(value)-required-optional)

def walk(value):
 if isinstance(value,dict):
  assert not set(value)&forbidden,('payload fields',set(value)&forbidden)
  for v in value.values():walk(v)
 elif isinstance(value,list):
  for v in value:walk(v)
 elif isinstance(value,str):
  assert not uid.search(value),'unredacted identifier'
  assert 'http://' not in value and 'https://' not in value,'raw URL'
 elif isinstance(value,float):assert math.isfinite(value)

def numeric(value):
 assert isinstance(value,(int,float)) and not isinstance(value,bool) and math.isfinite(value)

def observers(row):
 for g in row['geometry']:
  keys(g,{'time','parts'});numeric(g['time'])
  for p in g['parts']:
   keys(p,{'region','x','y','width','height'});assert p['region'] in regions
   for k in ['x','y','width','height']:numeric(p[k])
 for s in row['shifts']:
  keys(s,{'time','value','recentInput'}, {'sources'});assert isinstance(s['recentInput'],bool);numeric(s['time']);numeric(s['value'])
  for source in s.get('sources',[]):
   keys(source,{'previous','current'},{'tag'})
   if 'tag' in source:assert source['tag'] in {'::before','::after','A','ARTICLE','ASIDE','BODY','BUTTON','CANVAS','CODE','DD','DETAILS','DIALOG','DIV','DL','DT','EM','FIELDSET','FIGCAPTION','FIGURE','FOOTER','FORM','H1','H2','H3','H4','H5','H6','HEADER','HR','IMG','INPUT','LABEL','LI','MAIN','NAV','OL','OPTION','P','PRE','SECTION','SELECT','SMALL','SPAN','STRONG','SUMMARY','TABLE','TBODY','TD','TEXTAREA','TH','THEAD','TR','UL'}
   for r in [source['previous'],source['current']]:
    keys(r,rect_fields)
    for n in r.values():numeric(n)
 for t in row['longtasks']:keys(t,{'time','duration'});numeric(t['time']);numeric(t['duration']);assert t['duration']>=50
 for e in row['events']:keys(e,{'time','duration','interaction','name'});assert e['name'] in event_names;numeric(e['time']);numeric(e['duration']);numeric(e['interaction'])
 for s in row['scroll']:keys(s,{'time','y'});numeric(s['time']);numeric(s['y'])
 for l in row['lcp']:
  keys(l,{'time','size'},{'tag'});numeric(l['time']);numeric(l['size'])
  if 'tag' in l:assert l['tag'] in {'H1','H2','H3','P','SPAN','LI','DIV','IMG'}
