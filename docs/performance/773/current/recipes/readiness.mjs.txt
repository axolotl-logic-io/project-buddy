import puppeteer from 'puppeteer-core';
import {readFile,writeFile,mkdir} from 'node:fs/promises';
const s=JSON.parse(await readFile('/tmp/pb-773-session.json','utf8'));const base='http://localhost:36447';const output=process.cwd()+'/docs/performance/773/current';await mkdir(output,{recursive:true});
const browser=await puppeteer.launch({executablePath:'/home/bestieboots/.cache/ms-playwright/chromium-1248/chrome-linux64/chrome',headless:true,args:['--no-sandbox','--disable-dev-shm-usage']});
const journeys=[['home','/home','/projects'],['quests','/projects','/home'],['quest',`/projects/${s.project}`,'/projects'],['adventures','/chat','/home'],['long-adventure',`/chats/${s.chat}`,`/projects/${s.project}`],['activities','/activities','/home'],['workshop','/workshop','/home'],['editor',`/constructs/${s.construct}`,'/workshop']];
const rows=JSON.parse(await readFile(output+'/readiness.json','utf8'));const marker='audit-ready';
function route(v){return v.replace(base,'').replace(/[0-9a-f]{8}-[0-9a-f-]{27,}/g,':id').split('?')[0];}
async function observe(){
window.auditObservers?.forEach(o=>o.disconnect());window.auditObservers=[];
function watch(options,callback){const observer=new PerformanceObserver(callback);observer.observe(options);window.auditObservers.push(observer);}
window.audit={origin:performance.now(),shell:null,primary:null,geometry:[],scroll:[],shifts:[],longtasks:[],lcp:[],events:[]};let previous='';let live=true;window.stopAudit=()=>{live=false;};
const start=window.audit.origin;
watch({type:'layout-shift',buffered:true},list=>{for(const e of list.getEntries())if(e.startTime>=start)window.audit.shifts.push({time:e.startTime,value:e.value,recentInput:e.hadRecentInput,sources:e.sources?.map(x=>({tag:x.node?.tagName,previous:x.previousRect.toJSON(),current:x.currentRect.toJSON()}))});});
watch({type:'longtask',buffered:true},list=>{for(const e of list.getEntries())if(e.startTime>=start)window.audit.longtasks.push({time:e.startTime,duration:e.duration});});
watch({type:'largest-contentful-paint',buffered:true},list=>{for(const e of list.getEntries())if(e.startTime>=start)window.audit.lcp.push({time:e.startTime,size:e.size,tag:e.element?.tagName});});
watch({type:'event',durationThreshold:16,buffered:true},list=>{for(const e of list.getEntries())if(e.startTime>=start)window.audit.events.push({time:e.startTime,duration:e.duration,interaction:e.interactionId,name:e.name});});
function frame(){if(!live)return;const time=performance.now();const header=document.querySelector('body > header'),main=document.querySelector('main');if(!window.audit.shell&&header?.querySelector('a'))window.audit.shell=time;
const parts=[header,main,document.querySelector('main > .page-column'),...document.querySelectorAll('main textarea')].filter(Boolean).map((node,index)=>{const r=node.getBoundingClientRect();return {region:index===0?'header':index===1?'main':index===2?'column':'input',x:r.x,y:index===0?r.y:r.y+scrollY,width:r.width,height:r.height};});const value=JSON.stringify(parts);if(value!==previous){previous=value;window.audit.geometry.push({time,parts});}requestAnimationFrame(frame);}requestAnimationFrame(frame);window.addEventListener('scroll',()=>{if(live)window.audit.scroll.push({time:performance.now(),y:scrollY});},{passive:true});
}
async function visible(page,selector){for(const el of await page.$$(selector))if(await el.isVisible())return el;return null;}
async function probe(page,name){
const marker='audit-'+crypto.randomUUID();
let attempts=0;const start=performance.now();let firstAttempt;
while(performance.now()-start<15000){
 attempts++;
 if(firstAttempt===undefined)firstAttempt=await page.evaluate(()=>performance.now());
 if(name==='editor'||name==='long-adventure'){
  const selector=name==='editor'?'textarea[name="prompt"]':'textarea[aria-label="Message"]';
  const input=await visible(page,selector);if(!input)throw new Error('Expected input absent');await input.focus();await page.keyboard.type(marker);
 }else{
  const selector=name==='home'?'header button[aria-haspopup="dialog"]':name==='quests'?'button[role="tab"]:nth-child(2)':'main button[aria-haspopup="dialog"]';
  const control=await visible(page,selector);if(!control)throw new Error(`Expected control absent ${name}`);await control.click();
 }
 // Fixed 100 ms readiness probe resolution. This is lab sampling, not a behavior-test retry.
 await new Promise(r=>setTimeout(r,100));
 const success=await page.evaluate((name,marker)=>{
  if(name==='home')return !!document.querySelector('dialog[open]');
  if(name==='quests')return document.querySelectorAll('button[role="tab"]')[1]?.getAttribute('aria-selected')==='true';
  if(name==='editor')return Object.values(localStorage).some(v=>v.includes(marker));
  if(name==='long-adventure')return [...document.querySelectorAll('main button')].some(b=>b.textContent.trim()==='Send'&&!b.disabled);
  return !!document.querySelector('[role="dialog"] input,[role="dialog"] select');
 },name,marker);
 if(success){const confirmed=await page.evaluate(()=>performance.now());if(!['editor','long-adventure','quests'].includes(name))await page.keyboard.press('Escape');return {firstAttempt,confirmed,attempts,method:name==='home'?'menu-open':name==='quests'?'status-tab-state':name==='editor'?'persisted-draft':name==='long-adventure'?'send-enabled':'adventure-dialog-form'};}
}
throw new Error(`Control readiness not confirmed ${name}`);
}
try{
for(const state of ['populated','empty','empty-detail'])for(const profile of ['desktop','phone'])for(const [name,normalPath,normalSource] of journeys){
 if(state==='empty'&&!['home','quests','adventures','activities','workshop'].includes(name))continue;
 if(state==='empty-detail'&&!['quest','long-adventure'].includes(name))continue;
 const path=state==='empty-detail'?(name==='quest'?`/projects/${s.emptyProject}`:`/chats/${s.emptyChat}`):normalPath;
 const source=state==='empty-detail'?(name==='quest'?'/projects':`/projects/${s.emptyProject}`):normalSource;
 for(let sample=1;sample<=(state==='populated'?3:1);sample++){
 if(rows.filter(r=>r.state===state&&r.profile===profile&&r.name===name&&r.sample===sample).length===3)continue;
 const context=await browser.createBrowserContext();const page=await context.newPage();await page.setViewport(profile==='phone'?{width:390,height:844,isMobile:true,hasTouch:true,deviceScaleFactor:1}:{width:1440,height:900,deviceScaleFactor:1});await page.setCookie({name:'session',value:state==='empty-detail'?s.details:s[state],url:base,httpOnly:true,sameSite:'Lax'});const cdp=await page.createCDPSession();await cdp.send('Network.enable');await cdp.send('Network.emulateNetworkConditions',{offline:false,latency:profile==='phone'?150:0,downloadThroughput:profile==='phone'?200000:-1,uploadThroughput:profile==='phone'?93750:-1});await cdp.send('Emulation.setCPUThrottlingRate',{rate:profile==='phone'?4:1});await page.evaluateOnNewDocument(observe);
 for(const mode of ['cold-direct','warm-direct','client-navigation']){
 let navigation;
 if(mode==='client-navigation'){
  await page.goto(base+source,{waitUntil:'domcontentloaded'});await page.waitForFunction(()=>!document.querySelector('[aria-busy="true"]')&&document.querySelector('main')?.innerText.length>40);await page.evaluate(()=>window.stopAudit?.());await page.evaluate(observe);
  const anchor=await visible(page,`a[href="${path}"]`);if(!anchor)throw new Error('Missing route link');await anchor.click();await page.waitForFunction(path=>location.pathname===path,{},path);
 }else{const committed=new Promise(resolve=>{const handler=frame=>{if(frame===page.mainFrame()){page.off('framenavigated',handler);resolve();}};page.on('framenavigated',handler);});navigation=page.goto(base+path,{waitUntil:'domcontentloaded'});await committed;}
 if(page.url().split('?')[0]!==base+path)throw new Error('Intended signed-in page not reached');
 await page.waitForFunction(name=>{
  if(document.querySelector('[aria-busy="true"]'))return false;
  if(name==='home')return !!document.querySelector('main a[href="/projects"]');
  if(name==='editor')return !!document.querySelector('textarea[name="prompt"]');
  if(name==='long-adventure')return !!document.querySelector('textarea[aria-label="Message"]');
  if(name==='quests')return !!document.querySelector('button[role="tab"]');
  return !!document.querySelector('main button[aria-haspopup="dialog"]');
 },{},name);
 await page.evaluate(()=>window.audit.primary=performance.now());
 const control=await probe(page,name);
 // Observe at least five seconds after destination start and two seconds after confirmed control response.
 const remaining=await page.evaluate(confirmed=>Math.max(window.audit.origin+5000,confirmed+2000)-performance.now(),control.confirmed);if(remaining>0)await new Promise(r=>setTimeout(r,remaining));
 if(navigation){const response=await navigation;if(!response.ok())throw new Error(`Non-2xx ${response.status()}`);}
 const data=await page.evaluate(()=>{window.stopAudit();const n=performance.getEntriesByType('navigation')[0];return {...window.audit,end:performance.now(),nav:{responseStart:n.responseStart,requestStart:n.requestStart,start:n.startTime,dom:n.domContentLoadedEventEnd},resources:performance.getEntriesByType('resource').filter(r=>r.startTime>=window.audit.origin).map(r=>({path:r.name.replace(location.origin,'').replace(/[0-9a-f]{8}-[0-9a-f-]{27,}/g,':id').split('?')[0],start:r.startTime,duration:r.duration,bytes:r.transferSize,type:r.initiatorType}))};});
 rows.push({captureProtocol:'commit-probe-v2',state,profile,name,mode,sample,path:route(path),control,...data});await writeFile(output+'/readiness.json',JSON.stringify(rows));console.log(state,profile,name,sample,mode,'ready',Math.round(control.confirmed-data.origin),'attempts',control.attempts);
 }
 await context.close();
 }
}
}finally{await browser.close();}
