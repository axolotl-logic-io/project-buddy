import {readFile,writeFile} from 'node:fs/promises';import {createRequire} from 'node:module';import {setTimeout as delay} from 'node:timers/promises';import {loadavg} from 'node:os';
const require=createRequire(process.cwd()+'/package.json');const puppeteer=require('puppeteer-core');const f=JSON.parse(await readFile('/tmp/pb-current-diag/private.json','utf8'));const browser=await puppeteer.launch({executablePath:'/home/bestieboots/.cache/ms-playwright/chromium-1248/chrome-linux64/chrome',headless:true,args:['--no-sandbox','--disable-dev-shm-usage']});const rows=[];
try{for(let sample=1;sample<=10;sample++)for(const profile of ['desktop','phone'])for(const variant of (sample%2?['before','after']:['after','before'])){
const base='http://localhost:'+(Number(process.env.E2E_PORT)+(variant==='after'?1:0));const context=await browser.createBrowserContext();await context.setCookie({name:'session',value:f.session,url:base});const page=await context.newPage();await page.setViewport(profile==='desktop'?{width:1440,height:900}:{width:390,height:844,isMobile:true,hasTouch:true});
const cdp=await page.createCDPSession();await cdp.send('Network.enable');await cdp.send('Performance.enable');const pending=new Map();const network=[];
cdp.on('Network.requestWillBeSent',e=>{const u=new URL(e.request.url);if(u.origin!==base)return;let route='other';if(u.pathname==='/workshop')route='workshop-'+(u.searchParams.get('kind')??'construct');else if(u.pathname.startsWith('/_next/'))route='static';pending.set(e.requestId,{route,start:e.timestamp*1000,prefetch:Object.entries(e.request.headers).some(([k,v])=>/prefetch/i.test(k+' '+v)),initiator:e.initiator.type});});cdp.on('Network.loadingFinished',e=>{const r=pending.get(e.requestId);if(r)network.push({...r,end:e.timestamp*1000,bytes:e.encodedDataLength});});
await page.goto(base+'/workshop',{waitUntil:'domcontentloaded'});await page.waitForSelector('main a[aria-current="page"]');await page.focus('header button[aria-haspopup="dialog"]');await page.keyboard.press('Enter');await page.waitForSelector('dialog[open]');await page.keyboard.press('Escape');const closed=await page.evaluate(()=>performance.now());await delay(5000);
const now=await page.evaluate(()=>performance.now());const metrics=await cdp.send('Performance.getMetrics');const offset=metrics.metrics.find(m=>m.name==='Timestamp').value*1000-now;const captured=network.map(n=>({...n,start:n.start-offset,end:n.end-offset}));
const documentOrigin=await page.evaluate(()=>performance.timeOrigin);const started=await page.evaluate(()=>performance.now());await page.click('a[href="/workshop?kind=activity"]');await page.waitForSelector('a[aria-current="page"][href="/workshop?kind=activity"]');await page.waitForFunction(()=>document.querySelectorAll('main .definition-grid li').length>0);const arrival=await page.evaluate(()=>performance.now());if(await page.evaluate(()=>performance.timeOrigin)!==documentOrigin)throw Error('Full tab navigation');
rows.push({sample,variant,profile,closed,end:now,network:captured,tabConfirmation:arrival-started,load:loadavg()});await writeFile('/tmp/pb-current-diag/pair.json',JSON.stringify({protocol:'tab-prefetch-pair-v1',rows}));await context.close();
}}finally{await browser.close();}
