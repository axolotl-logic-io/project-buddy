import json,gzip,pathlib,statistics,collections
root=pathlib.Path('docs/performance/795')
rows=json.loads(pathlib.Path('/tmp/pb-795/geometry.json').read_text())
(root/'geometry.json.gz').write_bytes(gzip.compress(json.dumps(rows,separators=(',',':')).encode(),mtime=0))
lines=['# Paired geometry observations','','Production builds use the same synthetic accounts and database, browser, light theme, viewport, CDP throttling, source routes and content-presence +2 s observation window. Direct loads use fresh contexts; client navigation warms only the declared source before resetting observers immediately before activating its real link. Warm direct loads are outside this cohort.','','The direct metric is the largest initial-load session-window cumulative layout shift (CLS), excluding recent input. Transition all-shift sum includes recent-input shifts and is **not official CLS**. Values are median [min–max]; n is independent completed observations. Candidate limits are direct ≤0.1 and transition ≤0.05. All failed captures remain in the compressed archive.','','| Profile | Journey | Mode | Before n | Before | After n | After | After within candidate limit |','| --- | --- | --- | --- | --- | --- | --- | --- |']
for profile in ['desktop','phone']:
 for name in ['home','quests','quest','editor']:
  for mode in ['cold-direct','client-navigation']:
   metric='initialSessionWindowCls' if mode=='cold-direct' else 'transitionAllShiftSum'
   groups=[]
   for variant in ['before','after']:
    values=[r[metric] for r in rows if (r['variant'],r['profile'],r['name'],r['mode'])==(variant,profile,name,mode) and 'failure' not in r]
    groups.append(values)
   def fmt(values):
    return f'{statistics.median(values):.5f} [{min(values):.5f}–{max(values):.5f}]' if values else 'no completed sample'
   b,a=groups;limit=.1 if mode=='cold-direct' else .05
   lines.append(f'| {profile} | {name} | {mode} | {len(b)} | {fmt(b)} | {len(a)} | {fmt(a)} | {sum(x<=limit for x in a)}/{len(a)} |')
failures=[r for r in rows if 'failure' in r]
lines+=['',f'Capture failures: {len(failures)}.','', 'These observations establish finite-window lab behavior on this host, not field percentiles, interaction to next paint (INP), or permanent stability. Source rectangles and sampled geometry use document coordinates; scroll positions are retained separately. Activated-link safe-region rectangles are recorded independently from affected shift-source rectangles. No click-associated shift is excluded from transition totals.','']
(root/'results.md').write_text('\n'.join(lines))
# Numeric frames are privacy-safe rectangle filmstrips: region names and numeric geometry only.
frames=[{k:r[k] for k in ['variant','profile','name','mode','sample','geometry','scroll','shifts','activated','origin','primary','end']} for r in rows if 'failure' not in r and r['sample']==1]
(root/'geometry-filmstrips.json.gz').write_bytes(gzip.compress(json.dumps(frames,separators=(',',':')).encode(),mtime=0))
print('Summarized',len(rows),'observations;',len(failures),'failures')
# Six numeric geometry panels per journey; never a page screenshot outside Home.
film=root/'geometry-filmstrips';film.mkdir(exist_ok=True)
colors={'header':'#75604b','main':'#adc4d4','column':'#c27043','input':'#777777'}
for r in frames:
 width,height=(390,844) if r['profile']=='phone' else (1440,900)
 scale=210/width;panelHeight=height*scale;panelWidth=230
 times=[r['origin']+(r['end']-r['origin'])*i/5 for i in range(6)] if r['geometry'] else []
 pieces=[f'<svg xmlns="http://www.w3.org/2000/svg" width="{6*panelWidth}" height="{panelHeight+70}" viewBox="0 0 {6*panelWidth} {panelHeight+70}">','<rect width="100%" height="100%" fill="white"/>']
 for index,time in enumerate(times):
  sample=max((g for g in r['geometry'] if g['time']<=time),key=lambda g:g['time'],default=r['geometry'][0])
  offset=index*panelWidth
  pieces.append(f'<g transform="translate({offset+10},40)"><rect width="210" height="{panelHeight}" fill="none" stroke="#aaa"/><clipPath id="clip{index}"><rect width="210" height="{panelHeight}"/></clipPath><g clip-path="url(#clip{index})">')
  for p in sample['parts']:
   pieces.append(f'<rect x="{p["x"]*scale:.2f}" y="{p["y"]*scale:.2f}" width="{p["width"]*scale:.2f}" height="{p["height"]*scale:.2f}" fill="none" stroke="{colors[p["region"]]}" stroke-width="2"/>')
  pieces.append(f'</g><text y="-12" font-family="sans-serif" font-size="12">{time-r["origin"]:.0f} ms</text></g>')
 pieces.append('<text x="10" y="18" font-family="sans-serif" font-size="12">Numeric document geometry: header brown, main blue, column orange, input gray; scroll is separate.</text></svg>')
 (film/f'{r["variant"]}-{r["profile"]}-{r["name"]}-{r["mode"]}.svg').write_text(''.join(pieces))
