+// Local audit only. Each span owns its count; nested operations contribute
+// to their ancestors, while concurrent sibling operations remain separate.
+import { AsyncLocalStorage } from "node:async_hooks";
+import { z } from "zod";
+import { readServerSpans } from "../env.ts";
+import { logServerSpan } from "../errors/error-log.ts";
+
+const SERVER_OPERATIONS = z.enum([
+  "current-user",
+  "today",
+  "home-summary",
+  "workshop-options",
+  "construct-metadata",
+  "construct-data",
+  "activity-metadata",
+  "activity-data",
+  "spell-metadata",
+  "spell-data",
+]);
+
+type ServerOperation = z.infer<typeof SERVER_OPERATIONS>;
+
+interface ServerSpan {
+  readonly operation: ServerOperation;
+  readonly durationMs: number;
+  readonly queryCount: number;
+}
+
+interface QueryCounter {
+  count: number;
+}
+
+const counters = new AsyncLocalStorage<readonly QueryCounter[]>();
+
+// Counts executions, including failed queries. Drizzle may pass SQL and
+// parameters to its logger; this callback deliberately accepts neither.
+function countServerQuery(): void {
+  for (const counter of counters.getStore() ?? []) {
+    counter.count += 1;
+  }
+}
+
+// Preserves the operation's return value or error. Outputs only its static
+// name, elapsed milliseconds and inclusive query count. Counts are not
+// serial cost: simultaneous queries overlap within the elapsed duration.
+async function measureServerOperation<T>(
+  operation: ServerOperation,
+  work: () => Promise<T>,
+  now: () => number = performance.now.bind(performance),
+): Promise<T> {
+  if (!readServerSpans()) {
+    return work();
+  }
+  const name = SERVER_OPERATIONS.parse(operation);
+  const counter: QueryCounter = { count: 0 };
+  const start = now();
+  return counters.run([...(counters.getStore() ?? []), counter], async () => {
+    try {
+      return await work();
+    } finally {
+      logServerSpan({
+        operation: name,
+        durationMs: now() - start,
+        queryCount: counter.count,
+      });
+    }
+  });
+}
+
+export { countServerQuery, measureServerOperation };
+export type { ServerOperation, ServerSpan };
